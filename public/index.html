function textoNormalizado(v){
  return String(v || '')
    .trim()
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g,'')
    .replace(/\s+/g,' ');
}

function mismoTexto(a,b){
  return textoNormalizado(a) === textoNormalizado(b);
}

function rolActual(){
  return String(me?.role || '').toLowerCase();
}

function esUsuarioArea(){
  return ['operaciones','usuario_area','usuario'].includes(rolActual());
}

function usuarioVeSucursalArea(row){
  if(!me) return false;
  if(['admin','gerente','mantenimiento'].includes(rolActual())) return true;

  const sucUsuario = me.sucursal || '';
  const areaUsuario = me.area_asignada || me.area || '';
  const sucRow = row.sucursal || '';
  const areaRow = row.area || row.area_asignada || '';

  if(!sucUsuario || !areaUsuario) return false;
  return mismoTexto(sucUsuario, sucRow) && mismoTexto(areaUsuario, areaRow);
}

function activosVisibles(){
  return (activos || []).filter(usuarioVeSucursalArea);
}

function ticketsVisibles(){
  return (tickets || []).filter(usuarioVeSucursalArea);
}

function minutosBonitos(min){
  min = Number(min || 0);
  if(min < 60) return `${min} min`;
  const h = Math.floor(min / 60);
  const m = min % 60;
  if(h < 24) return m ? `${h} h ${m} min` : `${h} h`;
  const d = Math.floor(h / 24);
  const rh = h % 24;
  if(rh && m) return `${d} d ${rh} h ${m} min`;
  if(rh) return `${d} d ${rh} h`;
  return `${d} d`;
}

function kpiTiempo(titulo, min, detalle){
  return `<div class="kpi"><b>${minutosBonitos(min)}</b>${titulo}<br><span class="muted">${esc(detalle || (Number(min||0)+' min'))}</span></div>`;
}

assetSuggest = function(q){
  const box=document.getElementById('assetResults');
  if(!box) return;
  const nq=normalize(q);
  if(nq.length<1){ box.style.display='none'; box.innerHTML=''; return; }
  const rows=activosVisibles().filter(a=>normalize([
    a.numero,a.descripcion,a.area,a.tipo,a.sucursal,a.ubicacion,a.marca,a.modelo,a.usuario,a.estatus
  ].join(' ')).includes(nq)).slice(0,25);
  box.style.display='block';
  box.innerHTML=rows.length?rows.map(a=>`<div class="asset-option" onclick="selectAsset(${a.id})"><b>${esc(a.numero)}</b> · ${esc(a.descripcion)}<br><span class="muted">${esc(a.area)} · ${esc(a.tipo)} · ${esc(a.sucursal)} · ${esc(a.ubicacion)} · ${esc(a.usuario)}</span></div>`).join(''):`<div class="asset-option muted">Sin resultados para tu sucursal y área</div>`;
};

selectAsset = function(id){
  selectedAsset=activosVisibles().find(a=>String(a.id)===String(id));
  if(!selectedAsset){ alert('Este activo no pertenece a tu sucursal y área.'); return; }
  document.getElementById('assetResults').style.display='none';
  document.getElementById('assetQuery').value=selectedAsset.numero+' · '+selectedAsset.descripcion;
  document.getElementById('assetSelected').innerHTML=selectedAssetHtml(selectedAsset);
};

crearTicket = async function(){
  try{
    if(!selectedAsset){
      const q=val('assetQuery');
      const nq=normalize(q);
      selectedAsset = activosVisibles().find(a=>normalize([a.numero,a.descripcion].join(' ')).includes(nq));
    }
    if(!selectedAsset) return alert('Selecciona un activo desde la búsqueda');
    if(!val('falla').trim()) return alert('Describe la falla');
    showLoading('Creando ticket...');
    const fd=new FormData();
    fd.append('activo',selectedAsset.numero||'');
    fd.append('activo_descripcion',selectedAsset.descripcion||'');
    fd.append('area',selectedAsset.area || me.area_asignada || '');
    fd.append('sucursal',selectedAsset.sucursal || me.sucursal || 'SIN SUCURSAL');
    fd.append('ubicacion',selectedAsset.ubicacion||'');
    fd.append('solicitante',val('solicitante'));
    fd.append('empleado_solicitante',val('empSol'));
    fd.append('telefono_solicitante',val('telSol'));
    fd.append('falla',val('falla'));
    fd.append('prioridad',val('prioridad')||'Normal');
    fd.append('tipo_falla',val('tipoFalla'));
    filesOf('fotosReporte').forEach(f=>fd.append('fotos_reporte',f));
    const nuevo = await api('/api/tickets',{method:'POST',body:fd});
    selectedAsset=null;
    selectedReporter=null;
    await render();
    if(nuevo) pantallaAvisoGerente(nuevo);
  }catch(e){
    alert('Error al crear ticket: '+e.message);
  }finally{ hideLoading(); }
};

function camposUsuarioPorRol(){
  const role = val('eRole');
  const detalle = byId('camposUsuarioDetalle');
  const info = byId('ayudaUsuarioRol');
  if(!detalle || !info) return;
  if(role === 'operaciones'){
    detalle.style.display='none';
    info.innerHTML='Usuario de área: captura solo <b>Usuario</b>, <b>Sucursal</b> y <b>Área</b>. El nombre se llenará igual que el usuario. La comparación de área/sucursal no distingue mayúsculas, espacios ni acentos.';
  }else{
    detalle.style.display='grid';
    info.innerHTML='Personal de mantenimiento/gerencia: captura sus datos completos porque son usuarios independientes.';
  }
}

limpiarFormularioUsuario = function(){
  clearFields(['eNum','eUser','eName','ePass','eArea','eSuc','eTel','eMail']);
  const r=byId('eRole'); if(r) r.value='operaciones';
  camposUsuarioPorRol();
};

crearUsuario = async function(){
  try{
    showLoading('Creando usuario...');
    const role = val('eRole') || 'operaciones';
    const username = val('eUser').trim();
    const area = val('eArea').trim();
    const sucursal = val('eSuc').trim();
    if(!username) return alert('Escribe el usuario');
    if(!sucursal) return alert('Escribe la sucursal');
    if(!area) return alert('Escribe el área');
    const esArea = role === 'operaciones';
    await api('/api/users',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({
      numero_empleado: esArea ? '' : val('eNum'),
      username: username,
      name: esArea ? username : (val('eName') || username),
      password: val('ePass') || 'Temp1234',
      role: role,
      area_asignada: area,
      sucursal: sucursal,
      telefono: esArea ? '' : val('eTel'),
      correo: esArea ? '' : val('eMail')
    })});
    alert('Usuario creado. Tendrá que cambiar contraseña al entrar.');
    editingUserId=null;
    limpiarFormularioUsuario();
    await render();
  }catch(e){ alert('No se pudo crear usuario: '+e.message); }
  finally{ hideLoading(); }
};
